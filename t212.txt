#!/bin/sh
# Trading 212 - haalt portefeuille op en slaat het op naast dit bestand.
# Vul hieronder ALLEEN je key en secret in (tussen de aanhalingstekens).

KEY="PLAK_HIER_JE_API_KEY"
SECRET="PLAK_HIER_JE_API_SECRET"

# Resultaten komen in de map "output" naast dit bestand.
OUT="$(dirname "$0")/output"
BASE="https://live.trading212.com/api/v0/equity"
mkdir -p "$OUT"

curl -s -u "$KEY:$SECRET" "$BASE/account/summary"         > "$OUT/summary.json"
curl -s -u "$KEY:$SECRET" "$BASE/positions"               > "$OUT/positions.json"
curl -s -u "$KEY:$SECRET" "$BASE/history/orders?limit=50" > "$OUT/orders.json"

date > "$OUT/laatste_run.txt"
